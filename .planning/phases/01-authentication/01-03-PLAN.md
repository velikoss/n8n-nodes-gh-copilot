**Step 1: Get OAuth token via device code flow**

Run this Node.js script to get an OAuth token:

```javascript
// save as test-auth.mjs and run: node test-auth.mjs
import { requestDeviceCode, pollForToken } from './dist/lib/CopilotAuth.js';

async function main() {
  console.log('Requesting device code...');
  const deviceCode = await requestDeviceCode();

  console.log('\n=== AUTHORIZATION REQUIRED ===');
  console.log(`1. Go to: ${deviceCode.verification_uri}`);
  console.log(`2. Enter code: ${deviceCode.user_code}`);
  console.log('3. Authorize the application');
  console.log('==============================\n');

  console.log('Waiting for authorization...');
  const expiresAt = Date.now() + (deviceCode.expires_in * 1000);
  const token = await pollForToken(deviceCode.device_code, deviceCode.interval, expiresAt);

  console.log('\n=== SUCCESS ===');
  console.log(`OAuth Token: ${token}`);
  console.log('===============\n');
  console.log('Copy this token to use in n8n credentials.');
}

main().catch(console.error);
```

**Step 2: Test token exchange**

```javascript
// save as test-exchange.mjs and run: node test-exchange.mjs
import { getCopilotToken } from './dist/lib/CopilotTokenManager.js';

const oauthToken = 'YOUR_TOKEN_FROM_STEP_1';

async function main() {
  console.log('Exchanging token...');
  const response = await getCopilotToken(oauthToken);

  console.log('\n=== TOKEN EXCHANGE RESULT ===');
  console.log(`API Token: ${response.token.substring(0, 20)}...`);
  console.log(`Expires at: ${new Date(response.expires_at * 1000).toISOString()}`);
  console.log(`API Endpoint: ${response.endpoints?.api || 'default'}`);
  console.log('=============================\n');
}

main().catch(console.error);
```
